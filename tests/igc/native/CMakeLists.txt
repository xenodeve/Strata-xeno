cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(igc_output_native LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit 13.3 REQUIRED)
get_filename_component(IGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET igc_adapter)
  add_subdirectory("${IGC_ROOT}/tests/igc/host" host EXCLUDE_FROM_ALL)
endif()
add_library(igc_output STATIC "${IGC_ROOT}/src/kernels/igc_output.cu")
target_include_directories(igc_output PUBLIC "${IGC_ROOT}/include")
target_link_libraries(igc_output PUBLIC igc_adapter CUDA::cudart)
set_target_properties(igc_output PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
target_compile_options(igc_output PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(igc_output_probe "${IGC_ROOT}/tests/igc/output_probe.cpp")
target_compile_features(igc_output_probe PRIVATE cxx_std_20)
target_link_libraries(igc_output_probe PRIVATE igc_output)
add_executable(igc_fixture_probe "${IGC_ROOT}/tests/igc/fixture_probe.cpp")
target_compile_features(igc_fixture_probe PRIVATE cxx_std_20)
target_compile_definitions(igc_fixture_probe PRIVATE STRATA_IGC_NATIVE_DIAGNOSTIC=1)
target_link_libraries(igc_fixture_probe PRIVATE igc_output)
