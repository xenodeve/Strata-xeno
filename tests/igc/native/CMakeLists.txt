cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(igc_output_native LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit 13.3 REQUIRED)
get_filename_component(IGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET igc_adapter)
  add_subdirectory("${IGC_ROOT}/tests/igc/host" host EXCLUDE_FROM_ALL)
endif()
if(NOT TARGET igc_exact)
  add_library(igc_exact STATIC "${IGC_ROOT}/src/kernels/igc_exact.cu")
  target_include_directories(igc_exact PUBLIC "${IGC_ROOT}/include")
  target_link_libraries(igc_exact PUBLIC CUDA::cudart)
  set_target_properties(igc_exact PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
endif()
add_library(igc_output STATIC "${IGC_ROOT}/src/kernels/igc_output.cu")
target_include_directories(igc_output PUBLIC "${IGC_ROOT}/include")
target_link_libraries(igc_output PUBLIC igc_adapter igc_exact CUDA::cudart)
set_target_properties(igc_output PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
target_compile_options(igc_output PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(igc_output_probe "${IGC_ROOT}/tests/igc/output_probe.cpp")
target_compile_features(igc_output_probe PRIVATE cxx_std_20)
target_link_libraries(igc_output_probe PRIVATE igc_output)
add_executable(igc_fixture_probe "${IGC_ROOT}/tests/igc/fixture_probe.cpp")
target_compile_features(igc_fixture_probe PRIVATE cxx_std_20)
target_compile_definitions(igc_fixture_probe PRIVATE STRATA_IGC_NATIVE_DIAGNOSTIC=1)
target_link_libraries(igc_fixture_probe PRIVATE igc_output)
add_executable(igc_exact_broadcast "${IGC_ROOT}/tests/igc/exact_broadcast.cpp")
target_compile_features(igc_exact_broadcast PRIVATE cxx_std_20)
target_link_libraries(igc_exact_broadcast PRIVATE igc_exact)
add_library(igc_input STATIC "${IGC_ROOT}/src/kernels/igc_input.cu")
target_include_directories(igc_input PUBLIC "${IGC_ROOT}/include")
target_link_libraries(igc_input PUBLIC igc_adapter CUDA::cudart)
set_target_properties(igc_input PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
target_compile_options(igc_input PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(igc_input_probe "${IGC_ROOT}/tests/igc/input_probe.cpp")
target_compile_features(igc_input_probe PRIVATE cxx_std_20)
target_link_libraries(igc_input_probe PRIVATE igc_input)
add_library(igc_readiness STATIC "${IGC_ROOT}/src/kernels/igc_readiness.cu")
target_include_directories(igc_readiness PUBLIC "${IGC_ROOT}/include")
target_link_libraries(igc_readiness PUBLIC igc_adapter CUDA::cudart)
set_target_properties(igc_readiness PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
target_compile_options(igc_readiness PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(igc_readiness_probe "${IGC_ROOT}/tests/igc/readiness_probe.cpp")
target_compile_features(igc_readiness_probe PRIVATE cxx_std_20)
target_link_libraries(igc_readiness_probe PRIVATE igc_readiness)
add_library(igc_dispatch STATIC "${IGC_ROOT}/src/kernels/igc_dispatch.cu")
target_include_directories(igc_dispatch PUBLIC "${IGC_ROOT}/include")
target_link_libraries(igc_dispatch PUBLIC igc_exact CUDA::cudart)
set_target_properties(igc_dispatch PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_executable(igc_dispatch_test "${IGC_ROOT}/tests/igc/dispatch_test.cpp")
target_compile_features(igc_dispatch_test PRIVATE cxx_std_20)
target_link_libraries(igc_dispatch_test PRIVATE igc_dispatch)
add_executable(igc_invocation_test "${IGC_ROOT}/tests/igc/invocation_test.cpp")
target_compile_features(igc_invocation_test PRIVATE cxx_std_20)
target_link_libraries(igc_invocation_test PRIVATE igc_exact)
add_library(igc_context STATIC "${IGC_ROOT}/src/core/igc_context.cpp")
target_include_directories(igc_context PUBLIC "${IGC_ROOT}/include")
target_compile_features(igc_context PUBLIC cxx_std_20)
target_link_libraries(igc_context PUBLIC CUDA::cudart)
add_executable(igc_context_test "${IGC_ROOT}/tests/igc/context_test.cpp")
target_link_libraries(igc_context_test PRIVATE igc_context)
