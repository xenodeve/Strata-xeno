cmake_minimum_required(VERSION 3.24)
project(igc_adapter_host LANGUAGES CXX)
get_filename_component(IGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${IGC_ROOT}/tests/qwenputer/host" qwenputer-host EXCLUDE_FROM_ALL)
add_library(igc_adapter STATIC "${IGC_ROOT}/src/core/igc_adapter.cpp")
target_include_directories(igc_adapter PUBLIC "${IGC_ROOT}/include")
target_compile_features(igc_adapter PUBLIC cxx_std_20)
target_link_libraries(igc_adapter PUBLIC qwenputer_artifact)
if(MSVC)
  target_compile_options(igc_adapter PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(igc_adapter PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(igc_adapter_probe "${IGC_ROOT}/tests/igc/adapter_probe.cpp")
target_link_libraries(igc_adapter_probe PRIVATE igc_adapter)
