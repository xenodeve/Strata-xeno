cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(igc_adapter_host LANGUAGES CXX)
endif()
get_filename_component(IGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET qwenputer_artifact)
  add_subdirectory("${IGC_ROOT}/tests/qwenputer/host" qwenputer-host EXCLUDE_FROM_ALL)
endif()
add_library(igc_adapter STATIC "${IGC_ROOT}/src/core/igc_adapter.cpp")
target_include_directories(igc_adapter PUBLIC "${IGC_ROOT}/include")
target_compile_features(igc_adapter PUBLIC cxx_std_20)
target_link_libraries(igc_adapter PUBLIC qwenputer_artifact)
if(MSVC)
  target_compile_options(igc_adapter PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(igc_adapter PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(igc_adapter_probe "${IGC_ROOT}/tests/igc/adapter_probe.cpp")
target_link_libraries(igc_adapter_probe PRIVATE igc_adapter)
add_library(igc_controller_bundle STATIC "${IGC_ROOT}/src/core/igc_controller_bundle.cpp")
target_link_libraries(igc_controller_bundle PUBLIC igc_adapter)
if(MSVC)
  target_compile_options(igc_controller_bundle PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(igc_controller_bundle PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(igc_controller_bundle_probe "${IGC_ROOT}/tests/igc/controller_bundle_probe.cpp")
target_link_libraries(igc_controller_bundle_probe PRIVATE igc_controller_bundle)
add_executable(igc_inline_test "${IGC_ROOT}/tests/igc/inline_test.cpp")
target_include_directories(igc_inline_test PRIVATE "${IGC_ROOT}/include")
target_compile_features(igc_inline_test PRIVATE cxx_std_20)
add_executable(igc_coherent_observation_test "${IGC_ROOT}/tests/igc/coherent_observation_test.cpp")
target_compile_features(igc_coherent_observation_test PRIVATE cxx_std_20)
target_link_libraries(igc_coherent_observation_test PRIVATE qwenputer_artifact)
