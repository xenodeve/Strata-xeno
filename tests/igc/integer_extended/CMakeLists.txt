cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(igc_integer_extended LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit 13.3 REQUIRED)
get_filename_component(IGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(igc_integer_extended STATIC "${IGC_ROOT}/src/kernels/igc_integer_extended.cu")
target_include_directories(igc_integer_extended PUBLIC "${IGC_ROOT}/include")
target_compile_features(igc_integer_extended PUBLIC cxx_std_20)
target_link_libraries(igc_integer_extended PUBLIC CUDA::cudart)
set_target_properties(igc_integer_extended PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_executable(igc_integer_extended_test "${IGC_ROOT}/tests/igc/integer_extended_test.cpp")
if(NOT TARGET igc_exact)
  add_library(igc_exact STATIC "${IGC_ROOT}/src/kernels/igc_exact.cu")
  target_include_directories(igc_exact PUBLIC "${IGC_ROOT}/include")
  target_link_libraries(igc_exact PUBLIC CUDA::cudart)
  set_target_properties(igc_exact PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
endif()
# ABI1 is linked only into the test for rejection/unchanged-operation controls.
target_link_libraries(igc_integer_extended_test PRIVATE igc_integer_extended igc_exact)
