cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(qwenputer_host LANGUAGES CXX)
endif()
get_filename_component(_qwenputer_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(qwenputer_artifact STATIC "${_qwenputer_root}/src/core/qwenputer_artifact.cpp" "${_qwenputer_root}/src/core/qwenputer_call.cpp" "${_qwenputer_root}/src/core/qwenputer_binding.cpp")
target_include_directories(qwenputer_artifact PUBLIC "${_qwenputer_root}/include")
target_compile_features(qwenputer_artifact PUBLIC cxx_std_20)
if(WIN32)
  target_link_libraries(qwenputer_artifact PUBLIC bcrypt)
endif()
if(MSVC)
  target_compile_options(qwenputer_artifact PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(qwenputer_artifact PRIVATE -Wall -Wextra -Werror)
endif()
foreach(_test IN ITEMS artifact feature ownership call binding)
  add_executable(qwenputer_${_test}_test "${CMAKE_CURRENT_LIST_DIR}/../${_test}_test.cpp")
  target_link_libraries(qwenputer_${_test}_test PRIVATE qwenputer_artifact)
  if(MSVC)
    target_compile_options(qwenputer_${_test}_test PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(qwenputer_${_test}_test PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
add_executable(igc_preanchor_test "${CMAKE_CURRENT_LIST_DIR}/../../igc/preanchor_test.cpp")
target_include_directories(igc_preanchor_test PRIVATE "${_qwenputer_root}/include")
target_compile_features(igc_preanchor_test PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(igc_preanchor_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(igc_preanchor_test PRIVATE -Wall -Wextra -Werror)
endif()
