cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
  endif()
  project(qwenputer_native LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit 13.3 REQUIRED)
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET qwenputer_artifact)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../host" host)
endif()
add_library(qwenputer_cuda STATIC "${_root}/src/kernels/qwenputer.cu")
target_link_libraries(qwenputer_cuda PUBLIC qwenputer_artifact CUDA::cudart)
set_target_properties(qwenputer_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
target_compile_options(qwenputer_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(qwenputer_step_test "${CMAKE_CURRENT_LIST_DIR}/../step_test.cu")
target_link_libraries(qwenputer_step_test PRIVATE qwenputer_cuda)
set_target_properties(qwenputer_step_test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_executable(qwenputer_feature_gpu_test "${CMAKE_CURRENT_LIST_DIR}/../feature_gpu_test.cpp")
target_link_libraries(qwenputer_feature_gpu_test PRIVATE qwenputer_cuda)
add_library(qwenputer_runner STATIC "${_root}/src/kernels/qwenputer_runner.cu")
if(NOT TARGET strata_timeline)
  add_library(strata_timeline STATIC "${_root}/src/platform/timeline.cpp")
  target_include_directories(strata_timeline PUBLIC "${_root}/include")
  target_compile_features(strata_timeline PUBLIC cxx_std_17)
endif()
target_link_libraries(qwenputer_runner PUBLIC qwenputer_cuda strata_timeline)
set_target_properties(qwenputer_runner PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_executable(qwenputer_runner_test "${CMAKE_CURRENT_LIST_DIR}/../runner_test.cu")
target_link_libraries(qwenputer_runner_test PRIVATE qwenputer_runner)
set_target_properties(qwenputer_runner_test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_executable(qwenputer_run "${CMAKE_CURRENT_LIST_DIR}/../run.cpp")
target_link_libraries(qwenputer_run PRIVATE qwenputer_runner)
if(WIN32)
  target_link_libraries(qwenputer_run PRIVATE Psapi)
endif()
add_executable(qwenputer_state_test "${CMAKE_CURRENT_LIST_DIR}/../state_test.cu")
target_link_libraries(qwenputer_state_test PRIVATE qwenputer_cuda)
set_target_properties(qwenputer_state_test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
add_library(qwenputer_transaction STATIC "${_root}/src/core/qwenputer_transaction.cpp")
target_link_libraries(qwenputer_transaction PUBLIC qwenputer_runner)
target_compile_features(qwenputer_transaction PUBLIC cxx_std_20)
add_executable(qwenputer_transaction_test "${CMAKE_CURRENT_LIST_DIR}/../transaction_test.cpp")
target_link_libraries(qwenputer_transaction_test PRIVATE qwenputer_transaction)
